    41→/** reescala una imagen a dataURL (para wallpapers/avatares sin ocupar mucho localStorage) */
    42→function fileToDataUrl(file: File, maxW: number, quality = 0.82): Promise<string> {
    43→  return new Promise((resolve, reject) => {
    44→    const img = new Image();
    45→    const url = URL.createObjectURL(file);
    46→    img.onload = () => {
    47→      const scale = Math.min(1, maxW / img.width);
    48→      const canvas = document.createElement("canvas");
    49→      canvas.width = Math.round(img.width * scale);
    50→      canvas.height = Math.round(img.height * scale);
    51→      canvas.getContext("2d")?.drawImage(img, 0, 0, canvas.width, canvas.height);
    52→      URL.revokeObjectURL(url);
    53→      resolve(canvas.toDataURL("image/jpeg", quality));
    54→    };
    55→    img.onerror = () => {
    56→      URL.revokeObjectURL(url);
    57→      reject(new Error("No se pudo leer la imagen"));
    58→    };
    59→    img.src = url;
    60→  });
    61→}
